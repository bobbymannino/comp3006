% Extra LaTeX loaded into the preamble by build-pdf.sh
\usepackage{xcolor}
\usepackage{fancyhdr}

\definecolor{titleblue}{HTML}{1F365B}

% Footer with the page number on every page (fancy + plain, which
% LaTeX uses for chapter/TOC pages)
\fancypagestyle{plain}{%
  \fancyhf{}%
  \fancyfoot[C]{\thepage}%
  \renewcommand{\headrulewidth}{0pt}%
  \renewcommand{\footrulewidth}{0pt}%
}
\pagestyle{plain}

% Blue title page built from the title/author metadata
\makeatletter
\renewcommand{\maketitle}{%
  \begin{titlepage}
    \pagecolor{titleblue}\color{white}%
    \vspace*{\fill}
    \begin{center}
      {\Huge\bfseries \@title\par}
      \vspace{2em}
      {\Large Author: \@author\par}
      \vspace{1em}
      {\Large Updated: \today\par}
    \end{center}
    \vspace*{\fill}
  \end{titlepage}
  \nopagecolor
}
\makeatother

% Start the list of figures, list of tables and the body on fresh pages
\let\oldlistoffigures\listoffigures
\renewcommand{\listoffigures}{\clearpage\oldlistoffigures\clearpage}
\let\oldlistoftables\listoftables
\renewcommand{\listoftables}{\clearpage\oldlistoftables\clearpage}

% GitHub-style alerts (see pdf/alerts.lua): coloured bar + title
\usepackage{framed}
\definecolor{alertnote}{HTML}{0969DA}
\definecolor{alerttip}{HTML}{1A7F37}
\definecolor{alertimportant}{HTML}{8250DF}
\definecolor{alertwarning}{HTML}{9A6700}
\definecolor{alertcaution}{HTML}{CF222E}
\newenvironment{alert}[2]{%
  \def\FrameCommand{{\color{#1}\vrule width 3pt}\hspace{10pt}}%
  \MakeFramed{\advance\hsize-\width\FrameRestore}%
  \noindent{\color{#1}\bfseries #2}\par
}{\endMakeFramed}

% Keep figures exactly where they appear in the markdown
\usepackage{float}
\floatplacement{figure}{H}

% Zebra-striped table rows (needs xcolor's `table` option, passed via
% classoption in build-pdf.sh). longtable counts the caption and both header
% copies as rows, so start at row 4 to leave the header unshaded.
\definecolor{tablerow}{HTML}{F0F3F8}
\AtBeginEnvironment{longtable}{\rowcolors{4}{tablerow}{white}}
% Remove booktabs' padding around rules so the row colours have no gaps
\setlength{\aboverulesep}{0pt}
\setlength{\belowrulesep}{0pt}
\renewcommand{\arraystretch}{1.3}
